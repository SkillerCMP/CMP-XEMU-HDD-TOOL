cmake_minimum_required(VERSION 3.20)
project(XemuHddTools VERSION 0.6.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(XHC_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(XHC_BUILD_TESTS "Build regression programs" ON)
option(XHC_REQUIRE_QEMU_TESTS "Fail integration tests unless real qemu-img is present" OFF)
add_library(xhc STATIC src/common.cpp src/json.cpp src/process.cpp src/fatx.cpp src/converter.cpp src/csb/csb.cpp src/csb/custom-soundtrack-builder-core.cc src/hdd/hdd.cpp src/workspace.cpp)
target_include_directories(xhc PUBLIC include)
if(MSVC)
  target_compile_options(xhc PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(xhc PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(XHC_SANITIZERS AND NOT WIN32)
  target_compile_options(xhc PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(xhc PUBLIC -fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
target_link_libraries(xhc PUBLIC Threads::Threads)
add_executable(xemu-hdd-convert src/cli.cpp)
target_link_libraries(xemu-hdd-convert PRIVATE xhc)
if(NOT MSVC)
  target_compile_options(xemu-hdd-convert PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
  enable_language(RC)
  target_compile_definitions(xhc PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  add_executable(XEMU-HDD-Converter WIN32 src/gui_win32.cpp src/gui_csb_win32.cpp src/gui_splitter_win32.cpp src/gui_hdd_win32.cpp src/gui_workspace_win32.cpp src/gui_theme_win32.cpp src/app.rc)
  target_include_directories(XEMU-HDD-Converter PRIVATE src)
  set_target_properties(XEMU-HDD-Converter PROPERTIES OUTPUT_NAME "Xemu-HDD-Tools")
  target_sources(xemu-hdd-convert PRIVATE src/app.rc)
  target_include_directories(xemu-hdd-convert PRIVATE src)
  if(NOT MSVC)
    target_compile_options(XEMU-HDD-Converter PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  target_link_libraries(XEMU-HDD-Converter PRIVATE xhc comctl32 comdlg32 shell32 ole32 uuid gdi32 uxtheme dwmapi advapi32)
  if(MINGW)
    target_link_options(XEMU-HDD-Converter PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(xemu-hdd-convert PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(XHC_BUILD_TESTS)
  enable_testing()
  add_executable(xhc-theme-unit tests/theme_unit.cpp)
  target_link_libraries(xhc-theme-unit PRIVATE xhc)
  add_test(NAME theme-unit COMMAND xhc-theme-unit)
  add_executable(xhc-workspace-unit tests/workspace_unit.cpp)
  target_link_libraries(xhc-workspace-unit PRIVATE xhc)
  add_test(NAME workspace-unit COMMAND xhc-workspace-unit)
  add_executable(xhc-unit-tests tests/unit.cpp)
  target_link_libraries(xhc-unit-tests PRIVATE xhc)
  if(MINGW)
    target_link_options(xhc-unit-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME unit COMMAND xhc-unit-tests)
  add_executable(xhc-csb-unit tests/csb_unit.cpp)
  target_link_libraries(xhc-csb-unit PRIVATE xhc)
  add_test(NAME csb-unit COMMAND xhc-csb-unit)
  add_executable(xhc-hdd-unit tests/hdd_unit.cpp)
  target_link_libraries(xhc-hdd-unit PRIVATE xhc)
  add_test(NAME hdd-unit COMMAND xhc-hdd-unit)
  add_executable(xhc-hdd-driver tests/hdd_driver.cpp)
  target_link_libraries(xhc-hdd-driver PRIVATE xhc)
  add_executable(xhc-csb-driver tests/csb_driver.cpp)
  target_link_libraries(xhc-csb-driver PRIVATE xhc)
  add_executable(xhc-test12-oracle tests/reference_oracle.cpp tests/reference/fatx-hdd.cc)
  target_include_directories(xhc-test12-oracle PRIVATE tests/reference)
  target_link_libraries(xhc-test12-oracle PRIVATE xhc)
  if(MINGW)
    target_link_options(xhc-test12-oracle PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    if(NOT WIN32)
      add_executable(xhc-gui-theme-tests tests/gui_theme.cpp)
      target_include_directories(xhc-gui-theme-tests PRIVATE tests/win32-declarations src include)
      target_link_libraries(xhc-gui-theme-tests PRIVATE xhc)
      add_test(NAME gui-theme COMMAND xhc-gui-theme-tests)
      add_executable(xhc-gui-shell-tests tests/gui_shell.cpp)
      target_include_directories(xhc-gui-shell-tests PRIVATE tests/win32-declarations src)
      target_link_libraries(xhc-gui-shell-tests PRIVATE xhc)
      add_test(NAME gui-shell COMMAND xhc-gui-shell-tests)
      add_executable(xhc-gui-workspace-tests tests/gui_workspace.cpp)
      target_include_directories(xhc-gui-workspace-tests PRIVATE tests/win32-declarations src)
      target_link_libraries(xhc-gui-workspace-tests PRIVATE xhc)
      add_test(NAME gui-workspace COMMAND xhc-gui-workspace-tests)
      add_executable(xhc-gui-control-tests tests/gui_controls.cpp)
      target_include_directories(xhc-gui-control-tests PRIVATE tests/win32-declarations src)
      target_link_libraries(xhc-gui-control-tests PRIVATE xhc)
      add_test(NAME gui-controls COMMAND xhc-gui-control-tests)
      add_test(NAME gui-declaration-syntax COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/gui_syntax.py
        --compiler ${CMAKE_CXX_COMPILER})
      set_tests_properties(gui-declaration-syntax PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
    endif()
    add_test(NAME packaging COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/packaging.py)
    add_test(NAME build-timestamps COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/build_timestamps.py
      --cmake ${CMAKE_COMMAND} --compiler ${CMAKE_CXX_COMPILER})
    set_tests_properties(build-timestamps PROPERTIES TIMEOUT 240 SKIP_RETURN_CODE 77)
    if(XHC_SANITIZERS AND NOT WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      add_test(NAME sanitizer-toolchain COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/sanitizer_toolchain.py
        --compiler ${CMAKE_CXX_COMPILER})
      set_tests_properties(sanitizer-toolchain PROPERTIES TIMEOUT 240)
    endif()
    add_test(NAME csb-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/csb_integration.py
      --driver $<TARGET_FILE:xhc-csb-driver> --oracle $<TARGET_FILE:xhc-test12-oracle>)
    set_tests_properties(csb-integration PROPERTIES TIMEOUT 900)
    add_test(NAME hdd-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/hdd_integration.py
      --driver $<TARGET_FILE:xhc-hdd-driver> --oracle $<TARGET_FILE:xhc-test12-oracle>)
    set_tests_properties(hdd-integration PROPERTIES TIMEOUT 900)
    set(qemu_arg "")
    if(XHC_REQUIRE_QEMU_TESTS)
      set(qemu_arg --require-qemu)
    endif()
    add_test(NAME clean-conversions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py
      --binary $<TARGET_FILE:xemu-hdd-convert> --oracle $<TARGET_FILE:xhc-test12-oracle> ${qemu_arg})
    set_tests_properties(clean-conversions PROPERTIES TIMEOUT 900)
    if(NOT WIN32)
      add_executable(xhc-converter-faults tests/converter_faults.cpp)
      target_link_libraries(xhc-converter-faults PRIVATE xhc)
      add_test(NAME transaction-faults COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/faults.py
        --binary $<TARGET_FILE:xhc-converter-faults>)
      set_tests_properties(transaction-faults PROPERTIES TIMEOUT 300)
    endif()
  endif()
endif()
